function topProfile(scores:Record<string,number>){return Object.entries(scores).sort((a,b)=>b[1]-a[1]).slice(0,2).map(([k])=>dimensions[k as DimensionKey].label).join(' + ')}
function ProfileBars({scores}:{scores:Record<string,number>}){return <div className="profile-bars"><h4>{topProfile(scores)}</h4>{Object.entries(scores).sort((a,b)=>b[1]-a[1]).map(([k,v])=><div className="bar-row" key={k}><span>{dimensions[k as DimensionKey].label}</span><div className="bar"><i style={{width:`${Math.round(v/15*100)}%`}}/></div><strong>{v}/15</strong></div>)}</div>}
function CalendarLinks({appointment}:{appointment:Appointment}){if(!appointment.confirmed_start)return null;const start=new Date(appointment.confirmed_start);const end=new Date(start.getTime()+appointment.duration_minutes*60000);const stamp=(d:Date)=>d.toISOString().replace(/[-:]/g,'').replace(/\.\d{3}/,'');const title=encodeURIComponent(`Pathways counselling: ${appointment.topic}`);const dates=`${stamp(start)}/${stamp(end)}`;const google=`https://calendar.google.com/calendar/render?action=TEMPLATE&text=${title}&dates=${dates}`;const outlook=`https://outlook.office.com/calendar/0/deeplink/compose?subject=${title}&startdt=${encodeURIComponent(start.toISOString())}&enddt=${encodeURIComponent(end.toISOString())}`;return <div className="btn-row"><a className="secondary inline-button" href={google} target="_blank" rel="noreferrer">Add to Google Calendar</a><a className="secondary inline-button" href={outlook} target="_blank" rel="noreferrer">Add to Outlook</a></div>}
